1. Context Object Pattern (Recommended)
Create a shared context object that gets passed through the processor pipeline:

public class ProcessorContext {
    private Map<String, Object> data = new HashMap<>();
    private List<String> processedSteps = new ArrayList<>();
    private Map<String, Object> processorResults = new HashMap<>();
    
    public void put(String key, Object value) {
        data.put(key, value);
    }
    
    public Object get(String key) {
        return data.get(key);
    }
    
    public <T> T get(String key, Class<T> type) {
        return type.cast(data.get(key));
    }
    
    public void addProcessedStep(String processorName, Object result) {
        processedSteps.add(processorName);
        processorResults.put(processorName, result);
    }
    
    public List<String> getProcessedSteps() {
        return Collections.unmodifiableList(processedSteps);
    }
    
    public Map<String, Object> getProcessorResults() {
        return Collections.unmodifiableMap(processorResults);
    }
    
    public void clear() {
        data.clear();
        processedSteps.clear();
        processorResults.clear();
    }
}

3. Pipeline Engine with Context

public class ProcessorEngine {
    private List<Processor> processors = new ArrayList<>();
    private ProcessorContext context;
    
    public ProcessorEngine() {
        this.context = new ProcessorContext();
    }
    
    public void addProcessor(Processor processor) {
        processors.add(processor);
    }
    
    public ProcessorContext executePipeline(Object initialData) {
        context.clear();
        context.put("initialData", initialData);
        context.put("startTime", Instant.now());
        
        for (Processor processor : processors) {
            try {
                processor.process(context);
                context.addProcessedStep(processor.getName(), "SUCCESS");
            } catch (Exception e) {
                context.addProcessedStep(processor.getName(), "FAILED: " + e.getMessage());
                throw new PipelineException("Processor " + processor.getName() + " failed", e);
            }
        }
        
        context.put("endTime", Instant.now());
        context.put("status", "COMPLETED");
        
        return context;
    }
    
    public ProcessorContext getContext() {
        return context;
    }
}

4. Quartz Job Implementation

public class MyQuartzJob implements Job {
    private ProcessorEngine processorEngine;
    
    public MyQuartzJob() {
        this.processorEngine = new ProcessorEngine();
        // Initialize processors
        processorEngine.addProcessor(new DataValidationProcessor());
        processorEngine.addProcessor(new DataTransformationProcessor());
        processorEngine.addProcessor(new DataPersistenceProcessor());
    }
    
    @Override
    public void execute(JobExecutionContext quartzContext) throws JobExecutionException {
        try {
            // Get job data
            JobDataMap jobDataMap = quartzContext.getMergedJobDataMap();
            Object inputData = jobDataMap.get("inputData");
            
            // Execute pipeline
            ProcessorContext context = processorEngine.executePipeline(inputData);
            
            // Store results in quartz context for monitoring
            quartzContext.setResult(context.getProcessorResults());
            
            // Log processing summary
            logProcessingSummary(context);
            
        } catch (Exception e) {
            throw new JobExecutionException("Pipeline execution failed", e);
        }
    }
    
    private void logProcessingSummary(ProcessorContext context) {
        System.out.println("Processing Summary:");
        System.out.println("Total processors executed: " + context.getProcessedSteps().size());
        System.out.println("Processors: " + context.getProcessedSteps());
        System.out.println("Execution time: " + Duration.between(
            context.get("startTime", Instant.class),
            context.get("endTime", Instant.class)
        ).toMillis() + "ms");
    }
}

5. Enhanced Context with Metrics

public class EnhancedProcessorContext extends ProcessorContext {
    private Map<String, ProcessingMetrics> metrics = new HashMap<>();
    
    public void startProcessorTimer(String processorName) {
        metrics.put(processorName, new ProcessingMetrics(processorName));
    }
    
    public void stopProcessorTimer(String processorName) {
        ProcessingMetrics metric = metrics.get(processorName);
        if (metric != null) {
            metric.stop();
        }
    }
    
    public Map<String, ProcessingMetrics> getMetrics() {
        return Collections.unmodifiableMap(metrics);
    }
    
    public static class ProcessingMetrics {
        private String processorName;
        private Instant startTime;
        private Instant endTime;
        private String status;
        
        public ProcessingMetrics(String processorName) {
            this.processorName = processorName;
            this.startTime = Instant.now();
        }
        
        public void stop() {
            this.endTime = Instant.now();
        }
        
        public long getDurationMs() {
            return endTime != null ? Duration.between(startTime, endTime).toMillis() : 0;
        }
    }
}

6. Using the Context in Processors

public class DataTransformationProcessor implements Processor {
    @Override
    public void process(ProcessorContext context) {
        // Start timer
        if (context instanceof EnhancedProcessorContext) {
            ((EnhancedProcessorContext) context).startProcessorTimer(getName());
        }
        
        try {
            // Get data from previous processor
            boolean isValid = context.get("validationResult", Boolean.class);
            if (!isValid) {
                throw new ProcessorException("Data validation failed");
            }
            
            // Transformation logic
            Object transformedData = transformData(context.get("initialData"));
            
            // Store results
            context.put("transformedData", transformedData);
            context.addProcessedStep(getName(), transformedData);
            
        } finally {
            // Stop timer
            if (context instanceof EnhancedProcessorContext) {
                ((EnhancedProcessorContext) context).stopProcessorTimer(getName());
            }
        }
    }
    
    private Object transformData(Object data) {
        // Transformation logic
        return data;
    }
}

This approach gives you:

Complete visibility into what each processor accomplished

Execution metrics and timing information

Error tracking and status monitoring

Flexible data sharing between processors

Easy debugging and logging capabilities

The context object acts as a shared state that travels through your processor pipeline, collecting results and metadata along the way.
